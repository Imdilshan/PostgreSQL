```sql
select employee_name, company, salary
from employees
where category="Full-Time"
order by salary desc;
```

```sql
with fulltime as (
    select employee_name, company, salary
    from employees
    where category="Full-Time"
)
select employee_name, company, salary
from fulltime
order by salary desc;
```
